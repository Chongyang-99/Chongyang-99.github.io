        <button class="pj-arrow pj-arrow-prev" type="button" aria-label="Previous slide"><span aria-hidden="true">←</span></button>
        <button class="pj-arrow pj-arrow-next" type="button" aria-label="Next slide"><span aria-hidden="true">→</span></button>
        </div>
        </div>
        </div>
        </div>

        <script>
        (function() {
        'use strict';

        function initProjectDeck() {
        var deck = document.getElementById('pj-deck');
        if (!deck || deck.getAttribute('data-ready') === 'true') return;
        deck.setAttribute('data-ready', 'true');

        var deckConfig = [
        { chapter: 'Overview', exp: null },
        { chapter: 'Overview', exp: null },
        { chapter: 'Academic Research', exp: 'scu' },
        { chapter: 'Embodied Simulation', exp: 'megvii' },
        { chapter: 'ManiSkill-ViTac Track 1', exp: 'megvii' },
        { chapter: 'ManiSkill-ViTac Track 3', exp: 'megvii' },
        { chapter: 'RoboTwin Challenge', exp: 'dexmal' },
        { chapter: 'RoboTwin Challenge', exp: 'dexmal' },
        { chapter: 'Publications', exp: 'dexmal' },
        { chapter: 'Humanoid Loco-Manipulation', exp: 'tongyi' }
        ];
        var slides = Array.prototype.slice.call(deck.querySelectorAll('.pj-slide'));
        var thumbs = Array.prototype.slice.call(deck.querySelectorAll('.pj-thumb'));
        var expButtons = Array.prototype.slice.call(deck.querySelectorAll('.pj-exp-item'));
        var videoSlides = [];
        slides.forEach(function(slide, index) {
        var videos = slide.querySelectorAll('video');
        if (videos.length) videoSlides.push({ index: index, videos: videos });
        });
        var siteHeader = document.querySelector('nav[data-te-navbar-ref]');
        var documentRoot = document.documentElement;
        var canvas = deck.querySelector('.pj-canvas');
        var stageArea = deck.querySelector('.pj-stage-area');
        var thumbRail = deck.querySelector('.pj-thumbs');
        var previousButton = deck.querySelector('.pj-arrow-prev');
        var nextButton = deck.querySelector('.pj-arrow-next');
        var chapterChip = deck.querySelector('.pj-chapter-chip');
        var counter = deck.querySelector('.pj-counter span');
        var reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
        var current = 0;
        var transitioning = false;
        var transitionTarget = null;
        var pendingTarget = null;
        var touchStartX = 0;
        var touchStartY = 0;

        if (deckConfig.length !== slides.length || thumbs.length !== slides.length) {
        throw new Error('Project deck configuration mismatch');
        }

        function replayElements(slide) {
        slide.classList.remove('pj-replay');
        void slide.offsetWidth;
        slide.classList.add('pj-replay');
        }

        function preloadNearby(index, radius) {
        var preloadRadius = typeof radius === 'number' ? radius : 2;
        slides.forEach(function(slide, slideIndex) {
        if (Math.abs(slideIndex - index) > preloadRadius) return;
        Array.prototype.forEach.call(slide.querySelectorAll('img'), function(image) {
        image.loading = 'eager';
        if (typeof image.decode === 'function') image.decode().catch(function() {});
        });
        });
        }

        function syncLayout() {
        var headerHeight = siteHeader ? Math.ceil(siteHeader.getBoundingClientRect().height) : 0;
        var headerValue = headerHeight + 'px';
        if (documentRoot.style.getPropertyValue('--pj-header-height') !== headerValue) {
        documentRoot.style.setProperty('--pj-header-height', headerValue);
        }
        var canvasRect = canvas.getBoundingClientRect();
        var availableHeight = Math.max(0, window.innerHeight - canvasRect.top - 8);
        var areaStyle = window.getComputedStyle(stageArea);
        var sideGutters = (parseFloat(areaStyle.paddingLeft) || 0) + (parseFloat(areaStyle.paddingRight) || 0);
        var stageWidth = Math.max(0, Math.floor(Math.min(canvasRect.width - sideGutters, availableHeight * 16 / 9)));
        var stageValue = stageWidth + 'px';
        if (deck.style.getPropertyValue('--pj-stage-width') !== stageValue) {
        deck.style.setProperty('--pj-stage-width', stageValue);
        }
        }

        function syncSlideVideos(activeIndex) {
        videoSlides.forEach(function(entry) {
        Array.prototype.forEach.call(entry.videos, function(video) {
        if (entry.index === activeIndex) {
        video.play().catch(function() {});
        } else {
        video.pause();
        }
        });
        });
        }

        function updateControls(keepThumbVisible) {
        var activeConfig = deckConfig[current];
        chapterChip.textContent = activeConfig.chapter;
        counter.textContent = String(current + 1);
        previousButton.disabled = current === 0;
        nextButton.disabled = current === slides.length - 1;
        thumbs.forEach(function(thumb, index) {
        var active = index === current;
        thumb.classList.toggle('pj-current', active);
        if (active) thumb.setAttribute('aria-current', 'step');
        else thumb.removeAttribute('aria-current');
        });
        expButtons.forEach(function(button) {
        var active = button.getAttribute('data-exp') === activeConfig.exp;
        button.classList.toggle('pj-current', active);
        button.setAttribute('aria-pressed', active ? 'true' : 'false');
        });
        if (keepThumbVisible) {
        var railRect = thumbRail.getBoundingClientRect();
        var thumbRect = thumbs[current].getBoundingClientRect();
        if (thumbRect.top < railRect.top || thumbRect.bottom > railRect.bottom) {
        thumbRail.scrollTop += thumbRect.top - railRect.top - (thumbRail.clientHeight - thumbRect.height) / 2;
        }
        if (thumbRect.left < railRect.left || thumbRect.right > railRect.right) {
        thumbRail.scrollLeft += thumbRect.left - railRect.left - (thumbRail.clientWidth - thumbRect.width) / 2;
        }
        }
        preloadNearby(current);
        }

        function waitForTargetImages(target, callback) {
        var images = Array.prototype.slice.call(slides[target].querySelectorAll('img'));
        var pendingImages = images.filter(function(image) { return !image.complete; });
        if (!pendingImages.length) {
        callback();
        return;
        }
        var finished = false;
        var timer = window.setTimeout(finish, 600);
        function finish() {
        if (finished) return;
        finished = true;
        window.clearTimeout(timer);
        callback();
        }
        Promise.all(pendingImages.map(function(image) {
        image.loading = 'eager';
        return typeof image.decode === 'function' ? image.decode().catch(function() {}) : Promise.resolve();
        })).then(finish);
        }

        function goTo(target) {
        if (target < 0 || target >= slides.length) return;
        if (transitioning) {
        pendingTarget = target;
        return;
        }
        if (target === current) return;
        transitioning = true;
        transitionTarget = target;
        pendingTarget = null;
        preloadNearby(target, 1);
        waitForTargetImages(target, function() { performSlideTransition(target); });
        }

        function performSlideTransition(target) {
        var oldIndex = current;
        var direction = target > oldIndex ? 'next' : 'prev';
        var oldSlide = slides[oldIndex];
        var newSlide = slides[target];
        current = target;

        newSlide.classList.add('pj-active', 'pj-enter-' + direction);
        newSlide.setAttribute('aria-hidden', 'false');
        oldSlide.classList.add('pj-exit-' + direction);
        oldSlide.setAttribute('aria-hidden', 'true');
        replayElements(newSlide);
        syncSlideVideos(current);
        updateControls(true);

        window.setTimeout(function() {
        oldSlide.classList.remove('pj-active', 'pj-exit-' + direction, 'pj-replay');
        newSlide.classList.remove('pj-enter-' + direction);
        transitioning = false;
        transitionTarget = null;
        var queuedTarget = pendingTarget;
        pendingTarget = null;
        if (queuedTarget !== null && queuedTarget !== current) goTo(queuedTarget);
        }, reduceMotion.matches ? 30 : 470);
        }

        function stepBy(delta) {
        var base = pendingTarget !== null ? pendingTarget : (transitionTarget !== null ? transitionTarget : current);
        var target = Math.max(0, Math.min(slides.length - 1, base + delta));
        goTo(target);
        }

        function toggleFullscreen() {
        if (document.fullscreenElement === deck) {
        document.exitFullscreen();
        } else {
        deck.requestFullscreen();
        }
        }

        previousButton.addEventListener('click', function() { stepBy(-1); });
        nextButton.addEventListener('click', function() { stepBy(1); });
        thumbs.forEach(function(thumb) {
        thumb.addEventListener('click', function() {
        goTo(Number(thumb.getAttribute('data-slide')));
        });
        });
        expButtons.forEach(function(button) {
        button.addEventListener('click', function() {
        goTo(Number(button.getAttribute('data-slide')));
        });
        });

        document.addEventListener('keydown', function(event) {
        if (event.ctrlKey || event.metaKey || event.altKey) return;
        var targetElement = event.target;
        var targetTag = targetElement && targetElement.tagName;
        if (targetTag === 'INPUT' || targetTag === 'TEXTAREA' || targetTag === 'SELECT' || (targetElement && targetElement.isContentEditable)) return;
        if (event.key === 'ArrowLeft' || event.key === 'ArrowUp') {
        event.preventDefault();
        stepBy(-1);
        } else if (event.key === 'ArrowRight' || event.key === 'ArrowDown' || event.key === ' ') {
        event.preventDefault();
        stepBy(1);
        } else if (event.key.toLowerCase() === 'f') {
        event.preventDefault();
        toggleFullscreen();
        }
        });

        deck.addEventListener('touchstart', function(event) {
        if (event.touches.length !== 1) return;
        touchStartX = event.touches[0].clientX;
        touchStartY = event.touches[0].clientY;
        }, { passive: true });

        deck.addEventListener('touchend', function(event) {
        if (event.changedTouches.length !== 1) return;
        var deltaX = event.changedTouches[0].clientX - touchStartX;
        var deltaY = event.changedTouches[0].clientY - touchStartY;
        if (Math.abs(deltaX) < 45 || Math.abs(deltaX) <= Math.abs(deltaY) * 1.2) return;
        stepBy(deltaX < 0 ? 1 : -1);
        }, { passive: true });

        syncLayout();
        window.addEventListener('resize', syncLayout);
        if (typeof ResizeObserver === 'function') {
        var layoutObserver = new ResizeObserver(syncLayout);
        if (siteHeader) layoutObserver.observe(siteHeader);
        layoutObserver.observe(canvas);
        layoutObserver.observe(stageArea);
        }
        replayElements(slides[0]);
        syncSlideVideos(0);
        updateControls(false);
        }

        if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', initProjectDeck);
        } else {
        initProjectDeck();
        }
        })();
        </script>
